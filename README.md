## Python Program:

```
import requests
import json

class AIToolComparator:

    def __init__(self):
        self.results = {}

    def get_response(self, api_name, response_text):
        self.results[api_name] = response_text

    def compare_outputs(self):

        print("\n--- AI OUTPUT COMPARISON ---")

        for tool, output in self.results.items():
            print(f"\n{tool}:")
            print(output)

        outputs = list(self.results.values())

        if len(set(outputs)) == 1:
            print("\nResult: All AI tools generated similar outputs.")
        else:
            print("\nResult: Outputs differ among AI tools.")

    def generate_insights(self):

        print("\n--- ACTIONABLE INSIGHTS ---")

        longest = max(
            self.results,
            key=lambda x: len(self.results[x])
        )

        shortest = min(
            self.results,
            key=lambda x: len(self.results[x])
        )

        print(f"Most detailed response : {longest}")
        print(f"Shortest response      : {shortest}")

        for tool, output in self.results.items():
            print(f"{tool} response length = {len(output)}")

comparator = AIToolComparator()

comparator.get_response(
    "ChatGPT",
    "Machine Learning is a subset of AI that learns from data."
)

comparator.get_response(
    "Gemini",
    "Machine Learning enables systems to learn patterns from data."
)

comparator.get_response(
    "Claude",
    "Machine Learning uses algorithms to improve automatically through experience."
)

comparator.compare_outputs()
comparator.generate_insights()
```

## Optimized Python Program:
```
import requests

def get_ai_response(url, prompt):

    try:

        response = requests.post(
            url,
            json={"prompt": prompt},
            timeout=10
        )

        response.raise_for_status()

        return response.json()

    except requests.exceptions.Timeout:
        return {"error": "Request Timeout"}

    except requests.exceptions.RequestException as e:
        return {"error": str(e)}
```
